26:   function parseRegionsFromRows(rows){
27:     const header = rows[0] || [];
28:     const idxRegion = header.findIndex(h => /region|지역/i.test((h||'').trim()));
29:     const idxStaff = header.findIndex(h => /staff|담당/i.test((h||'').trim()));
30:     // identify school-name and school-code columns robustly
31:     let idxSchoolName = -1;
32:     let idxSchoolCode = -1;
33:     for (let i = 0; i < header.length; i++){
34:       const h = (header[i] || '').trim();
35:       const lower = h.toLowerCase();
36:       // detect explicit code columns first
37:       if (/(정보공시학교코드|학교코드|school_code|schoolcode|\bcode\b)/i.test(h)) {
38:         if (idxSchoolCode === -1) idxSchoolCode = i;
39:         continue;
40:       }
41:       // explicit name-like headers
42:       if (/(학교명|학교 이름|school_name|schoolname|school name|^school$)/i.test(h)) {
43:         idxSchoolName = i; break;
44:       }
45:     }
46:     // if explicit name header not found, pick any header containing '학교' or 'school' but not '코드'/'code'
47:     if (idxSchoolName === -1) {
48:       for (let i = 0; i < header.length; i++){
49:         const h = (header[i] || '').trim();
50:         if (/(학교|school)/i.test(h) && !/(코드|code)/i.test(h)) { idxSchoolName = i; break; }
51:       }
52:     }
53:     // final fallback: if we only found a code column, use it as code and don't treat it as name
54:     const idxSchool = idxSchoolName >= 0 ? idxSchoolName : idxSchoolCode;
55:     if (idxRegion < 0 || idxSchool < 0) return [];
56:     const regionMap = new Map();
57:     for (let i=1;i<rows.length;i++){
58:       const r = rows[i]; if(!r) continue;
59:       const rn = (r[idxRegion]||'').trim();
60:   const sn = idxSchoolName>=0 ? (r[idxSchoolName]||'').trim() : '';
61:   const scode = idxSchoolCode>=0 ? (r[idxSchoolCode]||'').trim() : '';
62:       const st = idxStaff>=0? (r[idxStaff]||'').trim() : '';
63:   if(!rn||!(sn||scode)) continue;
